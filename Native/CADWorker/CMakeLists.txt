cmake_minimum_required(VERSION 3.16)
project(PrevCADWorker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT APPLE OR NOT OCCT_BUILD)
  message(FATAL_ERROR "CADWorker requires macOS and the pinned OCCT_BUILD directory")
endif()
set(KERNEL_NAMES TKDESTEP TKDEIGES TKMesh TKXSBase TKShHealing TKBool TKBO TKPrim
  TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
set(KERNEL_LIBRARIES)
foreach(KERNEL IN LISTS KERNEL_NAMES)
  list(APPEND KERNEL_LIBRARIES "${OCCT_BUILD}/mac64/clang/lib/lib${KERNEL}.a")
endforeach()
add_executable(CADWorker main.cpp CADInput.cpp CADImport.cpp CADProtocol.cpp CADUnit.cpp)
target_include_directories(CADWorker PRIVATE "${OCCT_BUILD}/include/opencascade")
target_compile_options(CADWorker PRIVATE -fapplication-extension -Wall -Wextra -Werror=return-type)
target_link_options(CADWorker PRIVATE -Wl,-dead_strip)
target_link_libraries(CADWorker PRIVATE ${KERNEL_LIBRARIES})
add_executable(CADFixtureGenerator Tests/FixtureGenerator.cpp)
target_include_directories(CADFixtureGenerator PRIVATE "${OCCT_BUILD}/include/opencascade")
target_compile_options(CADFixtureGenerator PRIVATE -fapplication-extension)
target_link_options(CADFixtureGenerator PRIVATE -Wl,-dead_strip)
target_link_libraries(CADFixtureGenerator PRIVATE ${KERNEL_LIBRARIES})
# The protocol checks need no OCCT: build with `cmake --build <dir> --target CADProtocolTests`, or run
# Tests/test_protocol.py, which compiles the same files with the system C++ compiler. Never part of the default build.
add_executable(CADProtocolTests EXCLUDE_FROM_ALL Tests/ProtocolTests.cpp CADInput.cpp CADProtocol.cpp CADUnit.cpp)
target_include_directories(CADProtocolTests PRIVATE .)
